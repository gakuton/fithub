## 3. コンテキスト生成ロジック（`lib/utils/context.ts`）

### 3-1. 期間検出（`detectPeriod`）

ユーザーメッセージの文字列から参照期間を決定する。複数キーワードが存在する場合は最長期間を採用。

```ts
import { format, subDays, startOfMonth, endOfMonth, subMonths } from 'date-fns'

export type Period = { startDate: string; endDate: string }

type PeriodRule = { keywords: string[]; resolveFn: (today: Date) => Period }

const today = () => new Date()
const fmt   = (d: Date) => format(d, 'yyyy-MM-dd')

const PERIOD_RULES: PeriodRule[] = [
  {
    keywords:  ['1年', '今年', '12ヶ月', '12か月', '12カ月'],
    resolveFn: (t) => ({ startDate: fmt(subDays(t, 365)), endDate: fmt(t) }),
  },
  {
    keywords:  ['半年', '6ヶ月', '6か月', '6カ月'],
    resolveFn: (t) => ({ startDate: fmt(subDays(t, 180)), endDate: fmt(t) }),
  },
  {
    keywords:  ['3ヶ月', '3か月', '3カ月'],
    resolveFn: (t) => ({ startDate: fmt(subDays(t, 90)), endDate: fmt(t) }),
  },
  {
    keywords:  ['先月'],
    resolveFn: (t) => {
      const lastMonth = subMonths(t, 1)
      return { startDate: fmt(startOfMonth(lastMonth)), endDate: fmt(endOfMonth(lastMonth)) }
    },
  },
  {
    keywords:  ['今月', 'この1ヶ月', 'この1か月'],
    resolveFn: (t) => ({ startDate: fmt(subDays(t, 30)), endDate: fmt(t) }),
  },
  {
    keywords:  ['先週'],
    resolveFn: (t) => ({ startDate: fmt(subDays(t, 14)), endDate: fmt(subDays(t, 7)) }),
  },
  {
    keywords:  ['今週', 'この1週間', '1週間'],
    resolveFn: (t) => ({ startDate: fmt(subDays(t, 7)), endDate: fmt(t) }),
  },
  {
    keywords:  ['今日', '今日の'],
    resolveFn: (t) => ({ startDate: fmt(t), endDate: fmt(t) }),
  },
]

export function detectPeriod(message: string): Period {
  const t = today()
  // 最長期間のルールを優先（配列の先頭が最長）
  for (const rule of PERIOD_RULES) {
    if (rule.keywords.some((kw) => message.includes(kw))) {
      return rule.resolveFn(t)
    }
  }
  // デフォルト：直近 30 日
  return { startDate: fmt(subDays(t, 30)), endDate: fmt(t) }
}
```

### 3-3. コンテキスト文字列の組み立て（`buildChatContext`）

```ts
import { db } from '@/lib/db'
import { workoutSets, exercises, meals, mealItems, bodyCompositions, demographicData, motivations } from '@/lib/db/schema'
import { and, gte, lte, eq, desc, asc } from 'drizzle-orm'

export async function buildChatContext(
  period: Period,
  includesFoodDetail: boolean,
): Promise<string> {
  const sections: string[] = []

  // ── プロフィール（全件・期間フィルタなし） ──────────────
  const [demo] = await db.select().from(demographicData).limit(1)
  const goals  = await db.select().from(motivations).orderBy(desc(motivations.createdAt))

  if (demo) {
    const age = calcAge(demo.birthDate)
    const activityLabel = ACTIVITY_LEVEL_LABELS[demo.activityLevel] ?? demo.activityLevel
    sections.push(
      `【ユーザープロフィール】\n年齢: ${age}歳 / 性別: ${GENDER_LABELS[demo.gender] ?? demo.gender} / 身長: ${demo.heightCm}cm / 活動レベル: ${activityLabel}`,
    )
  }

  if (goals.length > 0) {
    const goalLines = goals.map((g) => {
      const cat   = MOTIVATION_LABELS[g.category] ?? g.category
      const state = g.achievedAt ? `（達成済み: ${g.achievedAt}）` : '（取り組み中）'
      return `  ${cat}: ${g.description}${state}`
    })
    sections.push(`【目標】\n${goalLines.join('\n')}`)
  }

  // ── 体組成履歴 ──────────────────────────────────────────
  const bodyRows = await db
    .select()
    .from(bodyCompositions)
    .where(and(gte(bodyCompositions.measuredDate, period.startDate), lte(bodyCompositions.measuredDate, period.endDate)))
    .orderBy(desc(bodyCompositions.measuredDate))

  if (bodyRows.length > 0) {
    const lines = bodyRows.map((r) => {
      const parts = [`体重 ${r.weightKg}kg`]
      if (r.bodyFatPct    != null) parts.push(`体脂肪率 ${r.bodyFatPct}%`)
      if (r.skeletalMuscleKg != null) parts.push(`骨格筋量 ${r.skeletalMuscleKg}kg`)
      if (r.bmr           != null) parts.push(`基礎代謝 ${r.bmr}kcal`)
      return `  ${r.measuredDate}: ${parts.join(' / ')}`
    })
    sections.push(`【体組成履歴（${period.startDate}〜${period.endDate}）】\n${lines.join('\n')}`)
  }

  // ── トレーニング履歴 ────────────────────────────────────
  const setRows = await db
    .select({ set: workoutSets, exerciseName: exercises.name })
    .from(workoutSets)
    .innerJoin(exercises, eq(workoutSets.exerciseId, exercises.id))
    .where(and(gte(workoutSets.workoutDate, period.startDate), lte(workoutSets.workoutDate, period.endDate)))
    .orderBy(asc(workoutSets.workoutDate), asc(workoutSets.setNumber))

  if (setRows.length > 0) {
    // 日付 → 種目 でグルーピング
    const byDate = new Map<string, Map<string, typeof setRows>>()
    for (const row of setRows) {
      if (!byDate.has(row.set.workoutDate)) byDate.set(row.set.workoutDate, new Map())
      const byEx = byDate.get(row.set.workoutDate)!
      if (!byEx.has(row.exerciseName)) byEx.set(row.exerciseName, [])
      byEx.get(row.exerciseName)!.push(row)
    }

    const dateLines: string[] = []
    for (const [date, byEx] of [...byDate].reverse()) {
      const exLines = [...byEx.entries()].map(([name, rows]) => {
        const sets = rows.map((r) => {
          const w = r.set.weightKg != null ? `${r.set.weightKg}kg×${r.set.reps}rep` : `自重×${r.set.reps}rep`
          const rm = r.set.estimated1rm != null ? `（推定1RM: ${r.set.estimated1rm}kg）` : ''
          return `${w}${rm}`
        })
        return `    ${name}: ${sets.join(', ')}`
      })
      dateLines.push(`  ${date}:\n${exLines.join('\n')}`)
    }
    sections.push(`【トレーニング履歴（${period.startDate}〜${period.endDate}）】\n${dateLines.join('\n')}`)
  }

  // ── 食事履歴 ────────────────────────────────────────────
  const mealRows = await db
    .select()
    .from(meals)
    .where(and(gte(meals.mealDate, period.startDate), lte(meals.mealDate, period.endDate)))
    .orderBy(desc(meals.mealDate))

  if (mealRows.length > 0) {
    if (includesFoodDetail) {
      // 詳細モード：食品名・食事タイプ単位で展開
      const mealLines: string[] = []
      for (const meal of mealRows) {
        const items = await db.select().from(mealItems).where(eq(mealItems.mealId, meal.id))
        const itemLines = items.map(
          (i) => `      ${i.foodName}: P${i.proteinG}g / F${i.fatG}g / C${i.carbG}g / ${i.kcal}kcal`,
        )
        mealLines.push(`  ${meal.mealDate} ${MEAL_TYPE_LABELS[meal.mealType] ?? meal.mealType}:\n${itemLines.join('\n')}`)
      }
      sections.push(`【食事記録・詳細（${period.startDate}〜${period.endDate}）】\n${mealLines.join('\n')}`)
    } else {
      // 集計モード（デフォルト）：日次 PFC 合計のみ
      const dailySummary = new Map<string, { p: number; f: number; c: number; kcal: number }>()
      for (const meal of mealRows) {
        const items = await db.select().from(mealItems).where(eq(mealItems.mealId, meal.id))
        const prev  = dailySummary.get(meal.mealDate) ?? { p: 0, f: 0, c: 0, kcal: 0 }
        for (const i of items) {
          prev.p    += i.proteinG
          prev.f    += i.fatG
          prev.c    += i.carbG
          prev.kcal += i.kcal
        }
        dailySummary.set(meal.mealDate, prev)
      }
      const summaryLines = [...dailySummary.entries()]
        .sort(([a], [b]) => b.localeCompare(a))
        .map(([date, s]) => `  ${date}: P${s.p.toFixed(1)}g / F${s.f.toFixed(1)}g / C${s.c.toFixed(1)}g / ${s.kcal.toFixed(0)}kcal`)
      sections.push(`【食事記録・日次集計（${period.startDate}〜${period.endDate}）】\n${summaryLines.join('\n')}`)
    }
  }

  return sections.join('\n\n')
}
```

## 4. セッション管理ユーティリティ（`lib/utils/chatSession.ts`）

```ts
export type ChatMessage = {
  role:      'user' | 'assistant'
  content:   string
  createdAt: string  // ISO 8601
}

type ChatSession = {
  date:     string          // YYYY-MM-DD（有効期限の判定キー）
  messages: ChatMessage[]
}

const SESSION_PREFIX = 'fithub_chat_'

/**
 * 午前 3 時を日付の境界とし、現在の「セッション日付」を返す。
 * 00:00〜02:59 の間は前日扱いとなる。
 */
function getSessionDate(): string {
  const now = new Date()
  if (now.getHours() < 3) {
    const yesterday = new Date(now)
    yesterday.setDate(yesterday.getDate() - 1)
    return formatDate(yesterday)
  }
  return formatDate(now)
}

function formatDate(d: Date): string {
  return d.toISOString().slice(0, 10)
}

export function loadSession(): ChatMessage[] {
  if (typeof window === 'undefined') return []

  const sessionDate = getSessionDate()

  // 当日以外のセッションキーを削除
  Object.keys(localStorage)
    .filter((key) => key.startsWith(SESSION_PREFIX) && key !== `${SESSION_PREFIX}${sessionDate}`)
    .forEach((key) => localStorage.removeItem(key))

  const raw = localStorage.getItem(`${SESSION_PREFIX}${sessionDate}`)
  if (!raw) return []

  try {
    const session: ChatSession = JSON.parse(raw)
    // 保存された日付が現在のセッション日付と一致するか確認
    if (session.date !== sessionDate) return []
    return session.messages
  } catch {
    return []
  }
}

export function saveSession(messages: ChatMessage[]): void {
  if (typeof window === 'undefined') return
  const sessionDate = getSessionDate()
  const session: ChatSession = { date: sessionDate, messages }
  localStorage.setItem(`${SESSION_PREFIX}${sessionDate}`, JSON.stringify(session))
}
```

## 5. APIルート（`app/api/ai/chat/route.ts`）

### エンドポイント

| メソッド | パス | 内容 |
|---------|------|------|
| POST | `/api/ai/chat` | メッセージを受け取りコンテキストを注入してストリーミング応答を返す |

### 処理フロー

```
POST /api/ai/chat
  ├─ 1. chatRequestSchema でバリデーション
  ├─ 2. 最後のユーザーメッセージから detectPeriod / detectFoodDetail
  ├─ 3. buildChatContext(period, includesFoodDetail)  ← DB クエリ
  ├─ 4. システムプロンプトを組み立て
  ├─ 5. Anthropic SDK client.messages.stream() でストリーミング開始
  └─ 6. ReadableStream として返却（Content-Type: text/plain; charset=utf-8）
```

### 実装

```ts
import Anthropic from '@anthropic-ai/sdk'
import { NextRequest } from 'next/server'
import { chatRequestSchema } from '@/lib/validations/chat'
import { detectPeriod, detectFoodDetail, buildChatContext } from '@/lib/utils/context'

const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY })

const SYSTEM_PROMPT_TEMPLATE = `\
あなたはFitHubユーザー専属のパーソナルトレーナーです。
以下のルールに従って回答してください。

【役割と行動指針】
- ユーザーのトレーニング・食事・体組成データを基に、具体的かつ実践的なアドバイスを提供する
- 発言には科学的根拠を伴わせる（必要に応じて研究・理論・計算根拠を示す）
- ですます調を使い、プロフェッショナルなトーンを維持する
- 過度な感情表現（「素晴らしい！」「最高です！」等）は控える
- データが不足している場合は、その旨を明示した上で回答する
- 回答は簡潔かつ実践的にまとめる

以下はユーザーの現在のフィットネスデータです：

{context}`

export async function POST(req: NextRequest): Promise<Response> {
  const body = await req.json().catch(() => null)
  const parsed = chatRequestSchema.safeParse(body)
  if (!parsed.success) {
    return Response.json({ error: 'Invalid request' }, { status: 400 })
  }

  const { messages } = parsed.data

  // 最後のユーザーメッセージでキーワード判定
  const lastUserMessage = [...messages].reverse().find((m) => m.role === 'user')
  const period          = detectPeriod(lastUserMessage?.content ?? '')
  const foodDetail      = detectFoodDetail(lastUserMessage?.content ?? '')

  let context: string
  try {
    context = await buildChatContext(period, foodDetail)
  } catch {
    return Response.json({ error: 'Failed to load context' }, { status: 500 })
  }

  const systemPrompt = SYSTEM_PROMPT_TEMPLATE.replace('{context}', context || 'データが登録されていません。')

  const encoder = new TextEncoder()
  const stream  = new ReadableStream({
    async start(controller) {
      try {
        const anthropicStream = client.messages.stream({
          model:      'claude-sonnet-4-6',
          max_tokens: 2048,
          system:     systemPrompt,
          messages,   // クライアントが直近 5 往復（10 メッセージ）に絞って送信済み
        })
        for await (const text of anthropicStream.textStream) {
          controller.enqueue(encoder.encode(text))
        }
      } catch {
        controller.enqueue(encoder.encode('\n\n（応答の取得に失敗しました。再送信してください。）'))
      } finally {
        controller.close()
      }
    },
  })

  return new Response(stream, {
    headers: { 'Content-Type': 'text/plain; charset=utf-8' },
  })
}
```

  6. 完了後 { role: 'assistant', content: streamingContent } を messages に追加
  7. saveSession(messages)、isLoading = false
  8. エラー時は error に文言をセット
